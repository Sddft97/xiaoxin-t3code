      <PopoverTrigger
        render={
          <ComposerControl
            ref={triggerRef}
            title={useChip ? triggerTooltipContent : undefined}
            data-stardust-model-chip={useChip || undefined}
            aria-description={useChip ? triggerTooltipContent : undefined}
            aria-label={[
              props.triggerAriaLabel ?? props.triggerLabel ?? allModelNames ?? triggerLabel,
              props.triggerDetail,
            ]
              .filter(Boolean)
              .join(", ")}

            data-chat-provider-model-picker="true"
            data-v9-unified-model-btn={props.isComposerOwned ? "true" : undefined}
            data-composer-shortcut={props.triggerShortcuts}
            onClick={(event) => {
              if (!isMenuOpen || !docked) return;
              const target = takeOpenTarget(triggerRef.current);
              if (!target) return;
              // A shortcut aimed at the open picker moves within it instead of closing it.
              (
                event as typeof event & { preventBaseUIHandler?: () => void }
              ).preventBaseUIHandler?.();
              openTarget(target);
            }}
            className={cn(
              "min-w-0 shrink justify-between whitespace-nowrap",
              props.compact
                ? "max-w-42 shrink-0"
                : !props.isComposerOwned && "max-w-48 sm:max-w-56",
              props.triggerClassName,
            )}
            disabled={props.disabled}
          />
        }
      >
        <span
          className={cn("flex min-w-0 flex-1 items-center", size === "xs" ? "gap-1" : "gap-1.5")}
        >
          {selectedEntries && props.triggerLabel === undefined ? (
            <span
              data-chat-provider-model-picker-icon
              className="flex shrink-0 items-center -space-x-1"
              aria-hidden="true"
            >
              {selectedEntries
                .slice(0, 3)
                .map((selection) =>
                  selection.entry ? (

                  +{selectedEntries.length - 3}
                </span>
              ) : null}
            </span>
          ) : activeEntry && props.triggerLabel === undefined ? (
            // Narrow compacted composers hide the provider icon before truncating the name.

              data-chat-provider-model-picker-icon
              data-v9-logo-chip={props.isComposerOwned ? "" : undefined}
              data-size={props.isComposerOwned ? "trigger" : undefined}
              className={props.isComposerOwned ? undefined : "contents"}

              {useChip && ChipIcon ? (
                <ChipIcon className="size-[15px]" />
              ) : (
                <ProviderInstanceIcon
